# US-0015: Tournament Club Picker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/admin/tournaments/[slug]/clubs` page where admins toggle which of the 15 master clubs are available in a tournament, defaulting to all-active.

**Architecture:** Server Component shell fetches clubs + existing `tournament_clubs` rows and passes merged state to a `ClubPickerForm` Client Component. Save calls `saveClubsAction` (delete-all-then-insert-all pattern) bound with `tournamentId` via `.bind()`. Tournament detail page gets a Clubs nav link.

**Tech Stack:** Next.js 14 App Router · TypeScript · Supabase (server client) · `useFormState` / `useFormStatus` · Vitest + React Testing Library

```typescript
'use server'

import { createClient } from '@/lib/supabase/server'

type ClubsActionState = { error: string | null }

// "no tournament_clubs rows" means all clubs are active — this invariant must
// be respected by any code that reads tournament_clubs (e.g. the bag picker).
export async function saveClubsAction(
  tournamentId: string,
  _prevState: ClubsActionState,
  formData: FormData
): Promise<ClubsActionState> {
  if (!tournamentId) return { error: 'tournamentId is required' }

  const supabase = createClient()

  const { data: isAdmin } = await supabase.rpc('fdgolf_is_admin')
  if (!isAdmin) return { error: 'Admin role required' }

  const activeClubIds = formData.getAll('active_club_id') as string[]

  const { data: clubs, error: clubsError } = await supabase
    .from('clubs')
    .select('id')

  if (clubsError || !clubs) {
    return { error: clubsError?.message ?? 'Failed to fetch clubs' }
  }

  const { error: deleteError } = await supabase
    .from('tournament_clubs')
    .delete()
    .eq('tournament_id', tournamentId)

  if (deleteError) return { error: deleteError.message }

  const rows = clubs.map((club) => ({
    tournament_id: tournamentId,
    club_id: club.id,
    is_active: activeClubIds.includes(club.id),
  }))

  const { error: insertError } = await supabase
    .from('tournament_clubs')
    .insert(rows)

  if (insertError) return { error: insertError.message }

  return { error: null }
}
```

```typescript
'use client'

import { useState } from 'react'
import { useFormState, useFormStatus } from 'react-dom'
import { saveClubsAction } from '@/lib/actions/clubs'
import { Button } from '@/components/ui/button'

interface Club {
  id: string
  display_name: string
  club_type: string
  display_order: number
}

interface Props {
  clubs: Club[]
  initialActiveIds: string[]
  tournamentId: string
}

const TYPE_LABELS: Record<string, string> = {
  wood: 'Woods',
  hybrid: 'Hybrids',
  iron: 'Irons',
  wedge: 'Wedges',
  putter: 'Putter',
}

const TYPE_ORDER = ['wood', 'hybrid', 'iron', 'wedge', 'putter']

function SaveButton() {
  const { pending } = useFormStatus()
  return (
    <Button type="submit" disabled={pending}>
      {pending ? 'Saving…' : 'Save'}
    </Button>
  )
}

export function ClubPickerForm({ clubs, initialActiveIds, tournamentId }: Props) {
  const [activeIds, setActiveIds] = useState<string[]>(initialActiveIds)
  const boundAction = saveClubsAction.bind(null, tournamentId)
  const [state, formAction] = useFormState(boundAction, { error: null })

  function toggle(id: string) {
    setActiveIds(prev =>
      prev.includes(id) ? prev.filter(x => x !== id) : [...prev, id]
    )
  }

  const grouped = TYPE_ORDER.reduce<Record<string, Club[]>>((acc, type) => {
    acc[type] = clubs.filter(c => c.club_type === type)
    return acc
  }, {})

  return (
    <div className="max-w-2xl mx-auto py-8 px-4 space-y-6">
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-bold">Available Clubs</h1>
        <div className="flex items-center gap-3">
          <span className="text-sm text-gray-500">
            {activeIds.length} of {clubs.length} active
          </span>
          <button
            type="button"
            onClick={() => setActiveIds(clubs.map(c => c.id))}
            className="text-xs px-2 py-1 border border-gray-300 rounded hover:bg-gray-50"
          >
            All
          </button>
          <button
            type="button"
            onClick={() => setActiveIds([])}
            className="text-xs px-2 py-1 border border-gray-300 rounded hover:bg-gray-50"
          >
            None
          </button>
        </div>
      </div>

      {state.error && (
        <p role="alert" className="text-sm text-red-600 bg-red-50 border border-red-200 rounded-md px-4 py-3">
          {state.error}
        </p>
      )}

      <form action={formAction} className="space-y-6">
        {/* Hidden inputs for active club IDs */}
        {activeIds.map(id => (
          <input key={id} type="hidden" name="active_club_id" value={id} />
        ))}

        {TYPE_ORDER.filter(type => grouped[type]?.length > 0).map(type => (
          <div key={type}>
            <p className="text-xs font-semibold text-gray-500 uppercase tracking-wide mb-2">
              {TYPE_LABELS[type]}
            </p>
            <div className="flex flex-wrap gap-2">
              {grouped[type].map(club => {
                const isActive = activeIds.includes(club.id)
                return (
                  <button
                    key={club.id}
                    type="button"
                    onClick={() => toggle(club.id)}
                    className={`px-3 py-1.5 rounded-full text-sm font-medium transition-colors ${
                      isActive
                        ? 'text-white'
                        : 'bg-gray-100 text-gray-400 line-through'
                    }`}
                    style={isActive ? { backgroundColor: '#0e2818' } : undefined}
                  >
                    {club.display_name}
                  </button>
                )
              })}
            </div>
          </div>
        ))}

        <SaveButton />
      </form>
    </div>
  )
}
```

import { notFound, redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { ClubPickerForm } from './club-picker-form'

interface PageProps {
  params: { slug: string }
}

export default async function ClubsPage({ params }: PageProps) {
  const supabase = createClient()

  const { data: isAdmin, error: adminError } = await supabase.rpc('fdgolf_is_admin')
  if (adminError || !isAdmin) redirect('/')

  const { data: tournament, error: tournamentError } = await supabase
    .from('tournaments')
    .select('id, name')
    .eq('slug', params.slug)
    .single()

  if (tournamentError || !tournament) notFound()

  const { data: clubs } = await supabase
    .from('clubs')
    .select('id, display_name, club_type, display_order')
    .order('display_order')

  const { data: tournamentClubs } = await supabase
    .from('tournament_clubs')
    .select('club_id, is_active')
    .eq('tournament_id', tournament.id)

  // "no rows" means all active — AC-0067 default
  const initialActiveIds =
    tournamentClubs && tournamentClubs.length > 0
      ? tournamentClubs.filter(r => r.is_active).map(r => r.club_id)
      : (clubs ?? []).map(c => c.id)

  return (
    <ClubPickerForm
      clubs={clubs ?? []}
      initialActiveIds={initialActiveIds}
      tournamentId={tournament.id}
    />
  )
}
